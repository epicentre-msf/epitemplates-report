% PDF preamble for epitemplates-report.
%
% Brand colours are NOT hard-coded here. scripts/inject-brand-colors.lua
% reads brand.yml and prepends the matching \definecolor{...}{HTML}{...}
% block to header-includes, which Quarto emits BEFORE this file. So the
% colour names used below (primary, secondary, secondarybg, impredprim,
% impredsec) — plus the quarto-callout-* colours — are already defined
% by the time LaTeX reaches the boxes and environments here.
%
% To change the palette, edit brand.yml (the single source of truth).
% css/_brand.scss still carries its own copy for the HTML / RevealJS
% SCSS layer (deferred to a later migration session), so brand.yml and
% css/_brand.scss must still be edited in lockstep until that lands —
% but this LaTeX file no longer duplicates any hex value.

\usepackage[framemethod=default]{mdframed}
\usepackage{mathpazo}
\usepackage{scrlayer-scrpage}

\newmdenv[skipabove=7pt,
skipbelow=7pt,
backgroundcolor=secondarybg,
innerleftmargin=5pt,
innerrightmargin=5pt,
innertopmargin=5pt,
leftmargin=0cm,
rightmargin=0cm,
rightline=false,
bottomline=false,
leftline=true,
topline=false,
linewidth=3pt,
linecolor=secondary,
innerbottommargin=5pt]{hintbox}

\newmdenv[skipabove=7pt,
skipbelow=7pt,
backgroundcolor=impredsec,
innerleftmargin=5pt,
innerrightmargin=5pt,
innertopmargin=5pt,
leftmargin=0cm,
rightmargin=0cm,
rightline=false,
bottomline=false,
leftline=true,
topline=false,
linewidth=3pt,
linecolor=impredprim,
innerbottommargin=5pt]{impbox}

\newenvironment{hint}{
\color{primary}
\begin{hintbox}}
{\end{hintbox}}

\newenvironment{important}{
\color{primary}
\begin{impbox}}
{\end{impbox}}

% Page-header logo. Override the path or the width per-document by
% adding to your YAML front matter:
%
%   format:
%     epitemplates-report-pdf:
%       header-includes:
%         - \def\pdflogo{img/your-logo.png}
%         - \def\pdflogowidth{4cm}
%
% header-includes is processed BEFORE this file, so \providecommand
% becomes a no-op when the user has already \def'd the macro.
\providecommand{\pdflogo}{img/epicentre_msf_logo.png}
\providecommand{\pdflogowidth}{5cm}

\cohead[{\includegraphics[width=\pdflogowidth]{\pdflogo}}]{}

% Brand the PDF body: colour all sectioning headings + the title and
% the hyperlinks with the primary brand colour. `primary` is defined
% by scripts/inject-brand-colors.lua (from brand.yml, or a consumer's
% brand), so the PDF carries the brand identity by default and
% re-brands when a consumer ships their own palette. `disposition`
% is the KOMA font shared by the title and every \section* level, so
% one \addtokomafont covers them all without dropping the bold/size.
\addtokomafont{disposition}{\color{primary}}
% Defer \hypersetup to \begin{document}: this header is injected before
% Quarto loads hyperref, so calling it now would be an undefined macro.
\AtBeginDocument{\hypersetup{linkcolor=primary, urlcolor=primary, citecolor=primary}}
